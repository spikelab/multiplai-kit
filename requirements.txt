# Dependencies for the shared workspace venv ($CLAUDE_MULTIPLAI_HOME/.venv),
# created by setup.sh. The kit's own registered hook (validate-syntax.sh) needs
# only pyyaml; the rest seed the venv that the marketplace skill plugins
# (multiplai-research/-dev/-media) run against.

# Kit hook
pyyaml                  # JSON/YAML validation in validate-syntax.sh

# Eval suite (see evals/README.md — `python -m pytest evals/`)
pytest

# Model access for the plugin skills (host-injected inside Claude Code; pinned
# here so the skills also run from the plain venv). >=0.2.139: ClaudeAgentOptions
# must carry `thinking`, which multiplai-core's run_agent forwards
# unconditionally — an older SDK raises TypeError on every call that sets it, and
# the plugin skills set it. The earlier >=0.2.116 reason still holds underneath:
# the hub's driver runner (multiplai-gui ADR 0002) needs the init system message
# to carry the session id — on 0.1.x it never does, so commissioned sessions hang
# at start and die on the init timeout. An already-provisioned venv volume
# holding an older release "satisfies" a bare requirement and never upgrades;
# editing this file triggers the re-sync, and the floor makes that sync actually
# upgrade.
claude-agent-sdk>=0.2.139

# Driver-mode runner (multiplai-gui hub, ADR 0002): WS server the hub proxies
# session frames through. >=14 for the modern asyncio API.
websockets>=14

# Skill helpers
yt-dlp                  # youtube-transcript skill (multiplai-media)
beautifulsoup4          # web content parsing (deep-research, extract-insights)

# Deep-research pipeline (multiplai-research)
httpx                   # async HTTP client for fetching pages
trafilatura             # article content extraction to markdown
tavily-python           # Tavily search API SDK (async)
exa-py                  # Exa search API SDK (async)
pydantic                # structured validation for LLM outputs and state
python-dotenv           # loads .env from project root for secrets
